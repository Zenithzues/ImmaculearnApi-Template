-- WARNING: This schema is for context only and is not meant to be run.
-- Table order and constraints may not be valid for execution.

CREATE TABLE public.chat_rooms (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name character varying,
  description text,
  type character varying DEFAULT 'direct'::character varying CHECK (type::text = ANY (ARRAY['direct'::character varying, 'group'::character varying, 'channel'::character varying]::text[])),
  created_by character varying NOT NULL,
  is_public boolean DEFAULT false,
  settings jsonb DEFAULT '{}'::jsonb,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  deleted_at timestamp with time zone,
  CONSTRAINT chat_rooms_pkey PRIMARY KEY (id),
  CONSTRAINT fk_chat_rooms_created_by FOREIGN KEY (created_by) REFERENCES public.users(id)
);
CREATE TABLE public.messages (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  room_id uuid NOT NULL,
  sender_id character varying NOT NULL,
  content text,
  type character varying DEFAULT 'text'::character varying CHECK (type::text = ANY (ARRAY['text'::character varying, 'image'::character varying, 'file'::character varying, 'system'::character varying, 'yjs_update'::character varying]::text[])),
  metadata jsonb DEFAULT '{}'::jsonb,
  yjs_awareness_state jsonb,
  reply_to uuid,
  deleted boolean DEFAULT false,
  deleted_at timestamp with time zone,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT messages_pkey PRIMARY KEY (id),
  CONSTRAINT fk_messages_room FOREIGN KEY (room_id) REFERENCES public.chat_rooms(id),
  CONSTRAINT fk_messages_sender FOREIGN KEY (sender_id) REFERENCES public.users(id)
);
CREATE TABLE public.room_participants (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  room_id uuid NOT NULL,
  user_id character varying NOT NULL,
  role character varying DEFAULT 'member'::character varying CHECK (role::text = ANY (ARRAY['member'::character varying, 'admin'::character varying, 'owner'::character varying]::text[])),
  joined_at timestamp with time zone DEFAULT now(),
  last_read_at timestamp with time zone DEFAULT now(),
  settings jsonb DEFAULT '{}'::jsonb,
  CONSTRAINT room_participants_pkey PRIMARY KEY (id),
  CONSTRAINT fk_room_participants_room FOREIGN KEY (room_id) REFERENCES public.chat_rooms(id),
  CONSTRAINT fk_room_participants_user FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.users (
  id character varying NOT NULL,
  email character varying UNIQUE,
  username character varying,
  role character varying,
  avatar_url text,
  status character varying,
  last_seen timestamp with time zone,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT users_pkey PRIMARY KEY (id)
);
